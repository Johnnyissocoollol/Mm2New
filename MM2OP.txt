local function setCharacterTransparencyLocal(transparency)
    local character = localplayer.Character
    if character then
        for _, part in pairs(character:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.LocalTransparencyModifier = transparency
            end
        end
    end
end

local function cleanupInvisibleChair()
    local inv = workspace:FindFirstChild('invischair')
    if inv then
        pcall(function() inv:Destroy() end)
    end
end

local function becomeInvisible()
    if not feInvisibleEnabled then return end
    
    local character = localplayer.Character
    if not character then return end
    
    setCharacterTransparencyLocal(invisTransparency)
    
    local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
    if humanoidRootPart then
        local savedpos = humanoidRootPart.CFrame
        
        pcall(function() character:MoveTo(seatTeleportPosition) end)
        task.wait(0.05)
        
        if not character:FindFirstChild("HumanoidRootPart") or character.HumanoidRootPart.Position.Y < voidLevelYThreshold then
            pcall(function() character:MoveTo(savedpos) end)
            return
        end
        
        local Seat = Instance.new('Seat')
        Seat.Parent = workspace
        Seat.Anchored = false
        Seat.CanCollide = false
        Seat.Name = 'invischair'
        Seat.Transparency = 1
        Seat.Position = seatTeleportPosition
        
        local Weld = Instance.new("Weld")
        Weld.Part0 = Seat
        
        local torso = character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
        if torso then
            Weld.Part1 = torso
            Weld.Parent = Seat
            task.wait()
            pcall(function() Seat.CFrame = savedpos end)
        else
            Seat:Destroy()
        end
    end
end

local flying = false
local flyMaxSpeed = 50
local flyAcceleration = 2
local flyCurrentSpeed = 0
local flyMoveDirection = Vector3.new()
local flyBodyGyro = nil
local flyBodyVelocity = nil
local flyConnection = nil

local function getCharacterParts()
    local character = localplayer.Character
    if not character then return nil, nil end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    return humanoid, rootPart
end

local function stopFly()
    if not flying then return end
    
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    if flyBodyGyro then
        flyBodyGyro:Destroy()
        flyBodyGyro = nil
    end
    if flyBodyVelocity then
        flyBodyVelocity:Destroy()
        flyBodyVelocity = nil
    end
    
    local humanoid, _ = getCharacterParts()
    if humanoid then
        humanoid.PlatformStand = false
    end
    
    flying = false
    flyCurrentSpeed = 0
end

local function updateFlying()
    local humanoid, rootPart = getCharacterParts()
    if not flying or not humanoid or humanoid.Health <= 0 or not rootPart then
        stopFly()
        return
    end
    
    local camera = workspace.CurrentCamera
    local moveDir = humanoid.MoveDirection
    
    if moveDir.Magnitude > 0.01 then
        flyCurrentSpeed = math.min(flyMaxSpeed, flyCurrentSpeed + flyAcceleration)
        flyMoveDirection = moveDir.Unit
    else
        flyCurrentSpeed = math.max(0, flyCurrentSpeed - flyAcceleration)
    end
    
    local horizontalVelocity = Vector3.new(flyMoveDirection.X, 0, flyMoveDirection.Z)
    local velocityVector = Vector3.zero
    
    if horizontalVelocity.Magnitude > 0 then
        velocityVector = horizontalVelocity.Unit * flyCurrentSpeed
    end
    
    local lookVector = camera.CFrame.LookVector.Unit
    local verticalFactor = flyMoveDirection:Dot(lookVector)
    local direction = verticalFactor < 0 and -1 or 1
    local horizontalLook = Vector3.new(lookVector.X, 0, lookVector.Z)
    
    if horizontalLook.Magnitude > 0 then
        horizontalLook = horizontalLook.Unit
    end
    
    local verticalStrength = math.abs(flyMoveDirection:Dot(horizontalLook))
    local verticalVelocity = lookVector.Y * direction * verticalStrength * flyCurrentSpeed
    
    flyBodyVelocity.Velocity = Vector3.new(velocityVector.X, verticalVelocity, velocityVector.Z)
    
    local speedPercent = (flyCurrentSpeed / flyMaxSpeed) * 30
    local angle = -math.rad(flyMoveDirection:Dot(lookVector) * speedPercent)
    flyBodyGyro.CFrame = CFrame.new(rootPart.Position, rootPart.Position + lookVector) * CFrame.Angles(angle, 0, 0)
end

local function startFly()
    if flying then
        stopFly()
        task.wait(0.1)
    end
    
    local humanoid, rootPart = getCharacterParts()
    if not humanoid or not rootPart then
        sendNotification("Fly", "Wait for character to load...")
        return
    end
    
    flying = true
    
    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.P = 100000
    flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyGyro.CFrame = rootPart.CFrame
    flyBodyGyro.Parent = rootPart
    
    flyBodyVelocity = Instance.new("BodyVelocity")
    flyBodyVelocity.P = 10000
    flyBodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyVelocity.Velocity = Vector3.new(0, 0, 0)
    flyBodyVelocity.Parent = rootPart
    
    humanoid.PlatformStand = true
    
    flyConnection = RunService.Heartbeat:Connect(updateFlying)
end

local function setFlySpeed(speed)
    if type(speed) == "number" and speed >= 0 then
        flyMaxSpeed = speed
    end
end

local function getRoleColor(player)
    if playerData and playerData[player.Name] then
        local role = playerData[player.Name].Role
        if role == "Murderer" then return Color3.fromRGB(255, 50, 50) end
        if role == "Sheriff" then return Color3.fromRGB(50, 150, 255) end
    end
    if player.Backpack:FindFirstChild("Knife") or (player.Character and player.Character:FindFirstChild("Knife")) then
        return Color3.fromRGB(255, 50, 50)
    elseif player.Backpack:FindFirstChild("Gun") or (player.Character and player.Character:FindFirstChild("Gun")) then
        return Color3.fromRGB(50, 150, 255)
    end
    return Color3.fromRGB(50, 255, 100)
end

        if not player.Parent then
            espHookedPlayers[player] = nil
        end
    end)

local function makePlayerGhost(player)
    if not player.Character then return end
    affectedPlayers[player] = true
    for _, part in ipairs(player.Character:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            part.Transparency = 1
            part.CanCollide = false
        end
    end
    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.CanCollide = false
        hrp.Transparency = 0.5
    end
end

local function restoreAllPlayers()
    for player in pairs(affectedPlayers) do
        if player.Character then
            for _, part in ipairs(player.Character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.Transparency = 0
                    part.CanCollide = true
                end
            end
        end
        affectedPlayers[player] = nil
    end
    affectedPlayers = {}
end

local function startKillAura()
    if killAuraRunning then return end
    if not killAuraEnabled then return end
    killAuraRunning = true
    killAuraLoop = task.spawn(function()
        while killAuraRunning and killAuraEnabled do
            task.wait(KILL_AURA_DELAY)
            if not isMurderer() then
                killAuraRunning = false
                restoreAllPlayers()
                break
            end
            local char = localplayer.Character
            if not char then continue end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            equipKnife()
            local knife = char:FindFirstChild("Knife")
            if not knife then continue end
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= localplayer and player.Character then
                    local targetHRP = player.Character:FindFirstChild("HumanoidRootPart")
                    local hum = player.Character:FindFirstChildOfClass("Humanoid")
                    if targetHRP and hum and hum.Health > 0 then
                        local dist = (targetHRP.Position - hrp.Position).Magnitude
                        if dist <= KILL_AURA_RANGE then
                            makePlayerGhost(player)
                            targetHRP.CFrame = hrp.CFrame + hrp.CFrame.LookVector * 2
                            if knife:FindFirstChild("Stab") then
                                pcall(function() knife.Stab:FireServer("Slash") end)
                            end
                        end
                    end
                end
            end
        end
        killAuraRunning = false
    end)
end

local function stopKillAura()
    killAuraEnabled = false
    killAuraRunning = false
    if killAuraLoop then task.cancel(killAuraLoop) end
    killAuraLoop = nil
    restoreAllPlayers()
end

local function getClosestPlayer()
    local closest = nil
    local shortestDistance = math.huge
    local char = localplayer.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= localplayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local distance = (hrp.Position - player.Character.HumanoidRootPart.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                closest = player
            end
        end
    end
    return closest
end

local function throwKnife()
    local character = localplayer.Character
    if not character then return false end
    if not isMurderer() then return false end
    local knife = character:FindFirstChild("Knife") or localplayer.Backpack:FindFirstChild("Knife")
    if not knife then return false end
    if knife.Parent ~= character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid:EquipTool(knife)
            task.wait(0.2)
        end
    end
    knife = character:FindFirstChild("Knife")
    if not knife then return false end
    local target = getClosestPlayer()

    if throwAuraRunning then return end
    if not throwAuraEnabled then return end

    throwAuraLoop = task.spawn(function()
        while throwAuraRunning and throwAuraEnabled do
            task.wait(0.3)
            if not isMurderer() then
                throwAuraRunning = false
                break
            end
            local char = localplayer.Character
            if not char then continue end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end

local function shootMurdererWithTeleport(savedPos)
    local char = localplayer.Character
    if not char then return false end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not humanoid or not hrp then return false end
    local gun = char:FindFirstChild("Gun") or localplayer.Backpack:FindFirstChild("Gun")
    if not gun then
        sendNotification("Error", "You don't have the gun!")
        return false
    end
    local murderer = findMurderer()
    if not murderer then return true end
    local murderHRP = murderer.Character and murderer.Character:FindFirstChild("HumanoidRootPart")
    if not murderHRP then return false end
    local teleportPos = murderHRP.CFrame * CFrame.new(0, 0, 3)
    hrp.CFrame = teleportPos
    if gun.Parent ~= char then
        humanoid:EquipTool(gun)
        task.wait(0.1)
    end
    local velocity = murderHRP.Velocity
    local predictedPos = murderHRP.Position + (velocity * 0.12)
    local args = { CFrame.new(char.RightHand.Position), CFrame.new(predictedPos) }
    if gun:FindFirstChild("Shoot") then
        pcall(function() gun.Shoot:FireServer(unpack(args)) end)
    end
    hrp.CFrame = savedPos
    task.delay(0.4, function() pcall(function() humanoid:UnequipTools() end) end)
    task.wait(0.3)
    return not isMurdererAlive()
end

local function killMurderWithRetry()
    if not isLocalSheriff() then
        sendNotification("Error", "You are not the sheriff!")
        return
    end
    local char = localplayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local savedPos = hrp.CFrame
    local killed = shootMurdererWithTeleport(savedPos)
    if killed then
        return
    end
    local attempts = 1
    local maxAttempts = 10
    while not killed and attempts < maxAttempts and isLocalSheriff() do
        attempts = attempts + 1
        for i = 5, 1, -1 do
            if not isLocalSheriff() then return end
            task.wait(1)
        end
        char = localplayer.Character
        if not char then return end
        hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        savedPos = hrp.CFrame
        if not isMurdererAlive() then killed = true break end
        killed = shootMurdererWithTeleport(savedPos)
    end
end

local function makeDraggable(button)
    local dragging = false
    local dragStart = nil
    local startPos = nil
    local activeTouch = nil
    local dragThreshold = 15
    local initialPressPos = nil
    local isTap = true
    local tapCheckConnection = nil

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or
           input.UserInputType == Enum.UserInputType.Touch then
            activeTouch = input
            initialPressPos = input.Position
            isTap = true
            dragging = false
            
            if tapCheckConnection then
                tapCheckConnection:Disconnect()
                tapCheckConnection = nil
            end
            
            tapCheckConnection = task.delay(0.1, function()
                if isTap and activeTouch == input then
                    dragging = false
                end
                tapCheckConnection = nil
            end)
        end
    end)

    button.InputChanged:Connect(function(input)
        if activeTouch and input.UserInputType == activeTouch.UserInputType then
            if input.UserInputType == Enum.UserInputType.Touch and input.Position then
                if initialPressPos and not dragging then
                    local delta = input.Position - initialPressPos
                    if math.abs(delta.X) > dragThreshold or math.abs(delta.Y) > dragThreshold then
                        isTap = false
                        dragging = true
                        dragStart = input.Position
                        startPos = button.Position
                        if tapCheckConnection then
                            tapCheckConnection:Disconnect()
                            tapCheckConnection = nil
                        end
                    end
                end
                
                if dragging and dragStart and startPos then
                    local delta = input.Position - dragStart
                    button.Position = UDim2.new(
                        startPos.X.Scale,
                        startPos.X.Offset + delta.X,
                        startPos.Y.Scale,
                        startPos.Y.Offset + delta.Y
                    )
                end
            end
        end
    end)

    button.InputEnded:Connect(function(input)
        if input == activeTouch then
            activeTouch = nil
            initialPressPos = nil
            dragStart = nil
            startPos = nil
            task.wait(0.05)
            dragging = false
            isTap = true
            if tapCheckConnection then
                tapCheckConnection:Disconnect()
                tapCheckConnection = nil
            end
        end
    end)
end

local function createCircleButton(parent, text, position, callback)
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(0, 70, 0, 70)
    button.Position = position
    button.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    button.BackgroundTransparency = 0.5

    return button

local function getActiveMap()
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Model") and obj:FindFirstChild("Spawns") and not obj.Name:lower():find("lobby") then
            return obj
        end
    end
    return nil
end

local function teleportToMap()
    local mapModel = getActiveMap()
    if not mapModel then 
        sendNotification("Teleport", "Could not find active map!")
        return false
    end
    local spawnsFolder = mapModel:FindFirstChild("Spawns")
    if spawnsFolder then
        local spawnPoints = spawnsFolder:GetChildren()
        if #spawnPoints > 0 then
            local randomSpawn = spawnPoints[math.random(1, #spawnPoints)]
            teleportTo(CFrame.new(randomSpawn.Position + Vector3.new(0, 3, 0)))
            return true
        end
    end
    if mapModel.PrimaryPart then
        teleportTo(mapModel.PrimaryPart.CFrame + Vector3.new(0, 5, 0))
        return true
    end
    sendNotification("Teleport", "Could not find spawn point!")
    return false

local function SkidFling(TargetPlayer)
    local Player = localplayer
    local Character = Player.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    
    if not TCharacter then 
        return 
    end

    local THumanoid  
    local TRootPart  
    local THead  
    local Accessory  
    local Handle  
    
    if TCharacter:FindFirstChildOfClass("Humanoid") then  
        THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")  
    end  
    if THumanoid and THumanoid.RootPart then  
        TRootPart = THumanoid.RootPart  
    end  
    if TCharacter:FindFirstChild("Head") then  
        THead = TCharacter.Head  
    end  
    if TCharacter:FindFirstChildOfClass("Accessory") then  
        Accessory = TCharacter:FindFirstChildOfClass("Accessory")  
    end  
    if Accessory and Accessory:FindFirstChild("Handle") then  
        Handle = Accessory.Handle  
    end  
    
    if Character and Humanoid and RootPart then  
        if RootPart.Velocity.Magnitude < 50 then  
            getgenv().OldPos = RootPart.CFrame  
        end  
        
        if THumanoid and THumanoid.Sit then  
            return 
        end  
        
        if THead then  
            workspace.CurrentCamera.CameraSubject = THead  
        elseif Handle then  
            workspace.CurrentCamera.CameraSubject = Handle  
        elseif THumanoid and TRootPart then  
            workspace.CurrentCamera.CameraSubject = THumanoid  
        end  
        
        if not TCharacter:FindFirstChildWhichIsA("BasePart") then  
            return  
        end  
        
        local FPos = function(BasePart, Pos, Ang)  
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang  
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)  
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)  
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)  
        end  
        
        local SFBasePart = function(BasePart)  
            local TimeToWait = 2  
            local Time = tick()  
            local Angle = 0  
            repeat  
                if RootPart and THumanoid then  
                    if BasePart.Velocity.Magnitude < 50 then  
                        Angle = Angle + 100  
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle),0 ,0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))  
                        task.wait()  
                    else  
                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))  
                        task.wait()  
                          
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))  
                        task.wait()  
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))  
                        task.wait()  
                    end  
                end  
            until Time + TimeToWait < tick()  
        end  
        
        workspace.FallenPartsDestroyHeight = 0/0  
        
        local BV = Instance.new("BodyVelocity")  
        BV.Parent = RootPart  
        BV.Velocity = Vector3.new(0, 0, 0)  
        BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)  
        
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)  
        
        if TRootPart then  
            SFBasePart(TRootPart)  
        elseif THead then  
            SFBasePart(THead)  
        elseif Handle then  
            SFBasePart(Handle)  
        else  
            return  
        end  
        
        BV:Destroy()  
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)  
        workspace.CurrentCamera.CameraSubject = Humanoid  
        
        if getgenv().OldPos then  
            repeat  
                RootPart.CFrame = getgenv().OldPos * CFrame.new(0, .5, 0)  
                Character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, .5, 0))  
                Humanoid:ChangeState("GettingUp")  
                for _, part in pairs(Character:GetChildren()) do  
                    if part:IsA("BasePart") then  
                        part.Velocity, part.RotVelocity = Vector3.new(), Vector3.new()  
                    end  
                end  
                task.wait()  
            until (RootPart.Position - getgenv().OldPos.p).Magnitude < 25  
        end  
    end
end

local function loadWeaponData()
    local success, data = pcall(function()
        return require(game:GetService("ReplicatedStorage").Database.Sync.Item)
    end)
    if success then Weapons = data end
    local success2, data2 = pcall(function()
        return require(game:GetService("ReplicatedStorage").Modules.ProfileData)
    end)
    if success2 then PlayerData = data2 end
end

local function ApplyInventory()
    RunService:UnbindFromRenderStep("SpawnWeapon")
    RunService:BindToRenderStep("SpawnWeapon", 0, function()
        if PlayerData and PlayerData.Weapons then
            PlayerData.Weapons.Owned = SpawnedInventory
        end
    end)
end

local function SpawnWeapon(name)
    if not Weapons or not PlayerData then
        loadWeaponData()
        task.wait(0.5)
    end
    if not OriginalInventory then
        OriginalInventory = {}
        if PlayerData and PlayerData.Weapons then
            for k, v in pairs(PlayerData.Weapons.Owned) do
                OriginalInventory[k] = v
            end
        end
    end
    local found = false
    if Weapons then
        for weaponName, _ in pairs(Weapons) do
            if weaponName:lower():find(name:lower()) then
                SpawnedInventory[weaponName] = 1
                found = true
            end
        end
    end
    if found then
        ApplyInventory()
        if localplayer.Character then localplayer.Character:BreakJoints() end
        task.spawn(function()
            localplayer.CharacterAdded:Wait()
            task.wait(0.2)
            ApplyInventory()
        end)
        sendNotification("Spawner", "Spawned: " .. name)
    else
        sendNotification("Spawner", "Weapon not found: " .. name)
    end
end

local function RestoreInventory()
    if OriginalInventory and PlayerData then
        RunService:UnbindFromRenderStep("SpawnWeapon")
        PlayerData.Weapons.Owned = OriginalInventory
        local backup = OriginalInventory
        OriginalInventory = nil
        SpawnedInventory = {}
        if localplayer.Character then localplayer.Character:BreakJoints() end
        task.spawn(function()
            localplayer.CharacterAdded:Wait()
            task.wait(0.2)
            if PlayerData then PlayerData.Weapons.Owned = backup end
        end)
        sendNotification("Spawner", "Inventory restored!")
    else
        sendNotification("Spawner", "No backup inventory found.")
    end
end
